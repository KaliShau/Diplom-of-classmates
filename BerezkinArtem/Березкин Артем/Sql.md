### Sql для создания таблиц и тестовые данные:
```SQL
-- Создание таблицы ролей
CREATE TABLE roles (
    id SERIAL PRIMARY KEY,
    name VARCHAR(30) UNIQUE NOT NULL,
    description TEXT
);

-- Создание таблицы пользователей
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    login VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    role_id INTEGER REFERENCES roles(id)
);

-- Создание таблицы персонала
CREATE TABLE staff (
    id SERIAL PRIMARY KEY,
    user_id INTEGER UNIQUE REFERENCES users(id),
    full_name VARCHAR(100) NOT NULL,
    position VARCHAR(50) NOT NULL,
    department VARCHAR(50) NOT NULL,
    phone VARCHAR(20),
    hire_date DATE NOT NULL,
    is_active BOOLEAN DEFAULT true
);

-- Создание таблицы инвентаря
CREATE TABLE inventory (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL,
    quantity INTEGER,
    unit VARCHAR(10),
    room VARCHAR(20),
    status VARCHAR(20)
);

-- Создание таблицы поставщиков
CREATE TABLE suppliers (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    contact VARCHAR(100),
    phone VARCHAR(20)
);

-- Создание таблицы закупок
CREATE TABLE purchases (
    id SERIAL PRIMARY KEY,
    supplier_id INTEGER REFERENCES suppliers(id),
    item_name VARCHAR(100) NOT NULL,
    quantity INTEGER,
    unit VARCHAR(10),
    price TEXT,
    status TEXT,
    date DATE NOT NULL
);

-- Создание таблицы заявок на ремонт
 CREATE TABLE requests ( 
	 id SERIAL PRIMARY KEY, 
	 inventory_id INTEGER REFERENCES inventory(id), 
	 user_id INTEGER REFERENCES users(id), 
	 date DATE NOT NULL, 
	 problem TEXT NOT NULL, 
	 status VARCHAR(20) DEFAULT 'Новая' 
 );

-- Таблица графика работы
CREATE TABLE work_schedule (
    id SERIAL PRIMARY KEY,
    zone TEXT NOT NULL,
    type TEXT NOT NULL,
    staff_id INTEGER REFERENCES staff(id),
    day DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL
);

-- Таблица транспортных средств
CREATE TABLE buses (
    id SERIAL PRIMARY KEY,
    model VARCHAR(100) NOT NULL,
    license_plate VARCHAR(20) UNIQUE NOT NULL,
    purchase_date DATE,
    last_tech_inspection DATE,
    next_tech_inspection DATE,
    status VARCHAR(20) DEFAULT 'Исправен',
    notes TEXT
);

-- Таблица типов запчастей (каталог всех возможных запчастей)
CREATE TABLE part_types (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    part_number VARCHAR(50) UNIQUE,
    description TEXT,
    compatible_bus_models TEXT
);

-- Таблица экземпляров запчастей с состоянием
CREATE TABLE bus_parts (
    id SERIAL PRIMARY KEY,
    part_type_id INTEGER REFERENCES part_types(id) NOT NULL,
    bus_id INTEGER REFERENCES buses(id) NOT NULL,
    serial_number VARCHAR(50),
    status VARCHAR(20) NOT NULL,
    quantity INTEGER,
    unit VARCHAR(10),
    installation_date DATE,
    defect_date DATE,
    defect_description TEXT
);

-- Таблица операций с запчастями (закупка, установка, списание)
CREATE TABLE part_operations (
    id SERIAL PRIMARY KEY,
    part_id INTEGER REFERENCES bus_parts(id),
    operation_type VARCHAR(20) NOT NULL,
    bus_id INTEGER REFERENCES buses(id),
    part_type_id INTEGER REFERENCES part_types(id),
    quantity INTEGER,
    unit VARCHAR(10),
    price TEXT,
    operation_date DATE NOT NULL,
    performed_by INTEGER REFERENCES staff(id),
    supplier_id INTEGER REFERENCES suppliers(id),
    notes TEXT
);

-- Роли
INSERT INTO roles (name, description) VALUES 
('Admin', 'Полный доступ'),
('Accountant', 'Доступ к основному функционалу');
```
